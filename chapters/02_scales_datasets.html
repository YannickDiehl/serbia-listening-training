2. Scales and Analysis Datasets

NoteAbout this chapter

The two datasets of all later chapters: one row per questionnaire (df) and one row per person (df_wide). Reverse-keyed items, the 23 preregistered scales and their scores, the three hypothesis composites, and the design variables of the models. Feeds the description of the measures in the Method section and Table 1.

source("R/packages.R")                  # all packages (Chapter 1)
source("R/settings.R")                  # waves, conditions, folders (Chapter 1)
x <- readRDS("cache/data_clean.rds")    # the analysis file (Chapter 1)

2.1 Items

Which items point the other way? Every item is answered from 0 to 10. For 22 items a high answer means less of the construct, such as “I feel isolated” for togetherness; they are turned around (10 − x), so that high always means more. The 24 need-satisfaction items alternate between six subscales and are named by them.

# Items: reverse-keyed items turned around, the need-satisfaction block named
# by its six subscales, and the design columns as factors

# the 22 reverse-keyed items: 10 - x on the scale from 0 to 10
reverse_items <- c(
  "Alienation1", "Alienation2", "Isolation1", "Isolation2",
  "Flourishing5", "Flourishing6",
  "Rogerian7", "Rogerian8", "Rogerian9",
  "ToleranceDiversity1",
  paste0("needSatisfaction", seq(2, 24, by = 2))
)

# old name = new name; the 24 need-satisfaction items rotate through the six
# subscales (item 1 autonomy satisfaction, 2 autonomy non-frustration, ...,
# 7 autonomy satisfaction again)
need <- \(numbers) paste0("needSatisfaction", numbers)
rename_map <- c(
  "Alienation1" = "NoAlienation1", "Alienation2" = "NoAlienation2",
  "Isolation1"  = "NoIsolation1",  "Isolation2"  = "NoIsolation2",
  set_names(paste0("AutonomySatisfaction",     1:4), need(c(1, 7, 13, 19))),
  set_names(paste0("AutonomyNonFrustrated",    1:4), need(c(2, 8, 14, 20))),
  set_names(paste0("RelatednessSatisfaction",  1:4), need(c(3, 9, 15, 21))),
  set_names(paste0("RelatednessNonFrustrated", 1:4), need(c(4, 10, 16, 22))),
  set_names(paste0("CompetenceSatisfaction",   1:4), need(c(5, 11, 17, 23))),
  set_names(paste0("CompetenceNonFrustrated",  1:4), need(c(6, 12, 18, 24)))
)

df <- x |>
  mutate(across(all_of(reverse_items), \(answer) 10 - answer)) |>
  rename_with(\(old) rename_map[old], all_of(names(rename_map))) |>
  mutate(
    Occasion  = factor(Occasion, levels = 0:3, labels = paste0("T", 0:3)),
    # Control first: group 1 in the multi-group models
    condition = factor(condition, levels = c("Control", "Experimental")),
    # "NO ANSW." and "." -> missing
    sex       = case_match(sex, "F" ~ "Female", "M" ~ "Male")
  )

2.3 Scale scores

How is a scale scored? As the mean of its items, if at least one is answered. The three hypothesis composites pool all items of their block into one mean, rather than averaging the scale scores, so that a scale with more items weighs more.

# Scale scores: the mean of a scale's items (at least one answered); the
# women's attitude score also with all nine items; and the three hypothesis
# composites, the mean of the pooled items of their block

scores <- scale_items |>
  map(\(items) row_means(df[setdiff(items, ceiling_items)], min_valid = 1))

composites <- list(H1_ListeningTogetherness = h1_items,
                   H2_CognitiveComplexity   = h2_items,
                   H3_Wellbeing             = h3_items) |>
  map(\(items) row_means(df[items], min_valid = 1))

df <- df |>
  bind_cols(scores) |>
  mutate(womenAttitude_full = row_means(pick(all_of(scale_items$womenAttitude)),
                                        min_valid = 1),
         .after = womenAttitude) |>
  bind_cols(composites)

2.4 Design variables

How do the models know who was trained when? Each municipality had one of ten trainers; each trainer trained one municipality of each condition. The codings of time and training, one line per condition and wave:

# Design variables: the trainer of each municipality (each trainer trained
# one municipality of each condition) and the codings of time and training
# that the multilevel models use (Chapter 10)

trainers <- tribble(
  ~trainer, ~trained_first, ~trained_second,
  1,        "Kula",         "Vršac",
  2,        "Valjevo",      "Bač",
  3,        "Sokobanja",    "Prijepolje",
  4,        "Leskovac",     "Arilje",
  5,        "Kruševac",     "Pirot",
  6,        "Šid",          "Požarevac",
  7,        "Raška",        "Šabac",
  8,        "NoviPazar",    "Lebane",
  9,        "Užice",        "Kikinda",
  10,       "Kanjiža",      "Vlasotince"
)

trainer_of <- trainers |>
  pivot_longer(c(trained_first, trained_second), values_to = "Municipality") |>
  transmute(Municipality, trainedBy = factor(trainer))

df <- df |>
  left_join(trainer_of, by = "Municipality") |>
  mutate(
    # 1 once the person's own group is trained: experimental from T1,
    # control at T3
    D = as.integer((condition == "Experimental" & Occasion != "T0") |
                   (condition == "Control"      & Occasion == "T3")),
    # piecewise time: the first training (from T1), the second training (T3)
    Time_Exp_Training     = ifelse(Occasion == "T0", 0, 1),
    Time_Control_Training = ifelse(Occasion == "T3", 1, 0),
    # Helmert contrasts: T0 against the later waves, T3 against the earlier
    TF_helmert = ifelse(Occasion == "T0", -3, 1),
    TS_helmert = ifelse(Occasion == "T3",  3, -1)
  )
df |>
  distinct(condition, Occasion, D, Time_Exp_Training, Time_Control_Training,
           TF_helmert, TS_helmert) |>
  arrange(condition, Occasion)

2.6 One row per person

What do the latent and the mediation models need? One row per person, with every wave’s answers side by side.

# One row per person: the items of every wave side by side (ListenOther1_T0,
# ...), age, sex and attendance, the centred covariates of the structured
# means models (Chapter 11), the composites per wave for the mediation models
# (Chapter 12), and one dummy per training round

# sex: the answer given most often (one person gave F and M once each; the
# tie goes to "Female", the first in alphabetical order)
modal_sex <- df |>
  drop_na(sex) |>
  count(id, sex) |>
  slice_max(n, n = 1, by = id, with_ties = FALSE) |>
  select(id, sex)

person <- df |>
  group_by(id) |>
  summarise(age     = mean(age, na.rm = TRUE),
            n_waves = n_distinct(Occasion),
            .groups = "drop") |>
  left_join(modal_sex, by = "id") |>
  mutate(age_c  = age - mean(age, na.rm = TRUE),   # centred at the mean
         female = case_match(sex, "Female" ~ 1, "Male" ~ 0))

df_wide <- df |>
  filter(!is.na(condition)) |>
  select(id, condition, Occasion, all_of(item_names)) |>
  pivot_wider(id_cols     = c(id, condition),
              names_from  = Occasion,
              values_from = all_of(item_names),
              names_glue  = "{.value}_{Occasion}") |>
  left_join(person, by = "id") |>
  left_join(distinct(df, id, Municipality), by = "id")

# the three composites per wave, named as the mediation models expect
mediation_composites <- df |>
  select(id, Occasion,
         Hypothesis1_ListeningTogetherness = H1_ListeningTogetherness,
         Hypothesis2_CognitiveComplexity   = H2_CognitiveComplexity,
         Hypothesis3_Wellbeing             = H3_Wellbeing) |>
  pivot_wider(id_cols     = id,
              names_from  = Occasion,
              names_glue  = "{Occasion}_{.value}",
              values_from = starts_with("Hypothesis"))

df_wide <- df_wide |>
  left_join(mediation_composites, by = "id") |>
  mutate(train_eg = ifelse(condition == "Experimental", 1, 0),  # trained T0-T1
         train_cg = ifelse(condition == "Control", 1, 0))       # trained T2-T3
